## modem / cellular
Run (root):
  dmesg | grep -iE 'modem|sipc|rmb|wwan|sim|imei'
  getprop | grep -iE 'modem|radio|imei'
  ls -l /vendor/lib/modules | grep -iE 'modem|sipc|rmb'
  find /vendor /odm /mnt/vendor -iname 'l_*.bin' -o -iname '*modem*.bin' 2>/dev/null
